cmake_minimum_required(VERSION 3.21)
project(hello-kcoreaddons LANGUAGES CXX)
find_package(ECM REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(KF6CoreAddons REQUIRED)
add_executable(hello-kcoreaddons main.cpp)
target_link_libraries(hello-kcoreaddons PRIVATE Qt6::Core KF6::CoreAddons)
install(TARGETS hello-kcoreaddons RUNTIME DESTINATION bin)
