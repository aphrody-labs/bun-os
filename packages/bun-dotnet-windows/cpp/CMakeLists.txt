# C++ half of @aphrody/bun-dotnet-windows, driven by ../scripts/build.ts (cmake -G Ninja, MSVC cl, Release):
#   detours        static, vendor/detours (aphrody-labs/Detours, MIT)
#   winrtact.dll   vendor/winget-cli UndockedRegFreeWinRT + WinGetServer manual activation client (MIT), unmodified
#   bunos_win_cpp  C ABI over DirectX 12 (vendor/directx-headers), WIL (vendor/wil) and C++/WinRT
cmake_minimum_required(VERSION 3.28)
project(bunos_win_cpp LANGUAGES C CXX)

if(NOT MSVC)
  message(FATAL_ERROR "bunos_win_cpp builds with MSVC (cl)")
endif()

set(BUNOS_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor" CACHE PATH "Bun OS vendor/ directory")
set(BUNOS_CPPWINRT "" CACHE PATH "C++/WinRT projection generated by cppwinrt.exe")
set(BUNOS_WINMD "" CACHE FILEPATH "Microsoft.Management.Deployment.winmd placed beside winrtact.dll")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")      # static CRT: no VC++ redistributable on the target
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)            # /GL + /LTCG
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 NDEBUG)
add_compile_options(/O2 /Oi /Gy /Zi /utf-8 /permissive- /Zc:__cplusplus /EHsc /guard:cf /arch:AVX2)
add_link_options(/DEBUG:FULL /OPT:REF /OPT:ICF /GUARD:CF /CETCOMPAT)

set(WINGET "${BUNOS_VENDOR}/winget-cli/src")
set(URFW "${WINGET}/Xlang/UndockedRegFreeWinRT/src/UndockedRegFreeWinRT/UndockedRegFreeWinRT")
set(GEN "${CMAKE_CURRENT_BINARY_DIR}/gen")
file(MAKE_DIRECTORY "${GEN}/detours")

# --- Detours --------------------------------------------------------------------------------------------------------
configure_file("${BUNOS_VENDOR}/detours/src/detours.h" "${GEN}/detours/detours.h" COPYONLY)
configure_file("${BUNOS_VENDOR}/detours/src/detver.h" "${GEN}/detours/detver.h" COPYONLY)
add_library(detours STATIC
  "${BUNOS_VENDOR}/detours/src/detours.cpp"
  "${BUNOS_VENDOR}/detours/src/modules.cpp"
  "${BUNOS_VENDOR}/detours/src/disasm.cpp"
  "${BUNOS_VENDOR}/detours/src/image.cpp"
  "${BUNOS_VENDOR}/detours/src/creatwth.cpp"
  "${BUNOS_VENDOR}/detours/src/disolx86.cpp"
  "${BUNOS_VENDOR}/detours/src/disolx64.cpp"
  "${BUNOS_VENDOR}/detours/src/disolia64.cpp"
  "${BUNOS_VENDOR}/detours/src/disolarm.cpp"
  "${BUNOS_VENDOR}/detours/src/disolarm64.cpp")
target_include_directories(detours PRIVATE "${BUNOS_VENDOR}/detours/src")
target_compile_definitions(detours PRIVATE DETOUR_DEBUG=0)
set_target_properties(detours PROPERTIES INTERPROCEDURAL_OPTIMIZATION OFF)

# --- winrtact.dll ---------------------------------------------------------------------------------------------------
add_custom_command(
  OUTPUT "${GEN}/WinGetServer.h" "${GEN}/WinGetServer_c.c"
  COMMAND midl.exe /nologo /app_config /h WinGetServer.h /client stub /server none /out "${GEN}" "${WINGET}/WinGetServer/WinGetServer.idl"
  DEPENDS "${WINGET}/WinGetServer/WinGetServer.idl"
  COMMENT "midl WinGetServer.idl (manual activation RPC client)")
add_library(winrtact SHARED
  "${URFW}/dllmain.cpp"
  "${URFW}/catalog.cpp"
  "${URFW}/typeresolution.cpp"
  "${WINGET}/WinGetServer/WinGetServerManualActivation_Client.cpp"
  "${WINGET}/WinGetServer/Utils.cpp"
  "${GEN}/WinGetServer_c.c"
  "${URFW}/winrtact.def")
target_include_directories(winrtact PRIVATE "${GEN}" "${URFW}" "${WINGET}/WinGetServer" "${BUNOS_VENDOR}/wil/include")
target_compile_definitions(winrtact PRIVATE UNDOCKEDREGFREEWINRT_EXPORTS _WINDOWS _USRDLL)
target_compile_options(winrtact PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/std:c++17> /W3)
target_link_libraries(winrtact PRIVATE detours comsuppw shlwapi xmllite runtimeobject pathcch rometadata rpcrt4 shell32 advapi32 ole32)
if(BUNOS_WINMD)
  add_custom_command(TARGET winrtact POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${BUNOS_WINMD}" "$<TARGET_FILE_DIR:winrtact>/Microsoft.Management.Deployment.winmd")
endif()

# --- bunos_win_cpp.dll ----------------------------------------------------------------------------------------------
add_library(bunos_win_cpp SHARED bunos_win_cpp.cpp)
target_include_directories(bunos_win_cpp PRIVATE
  "${BUNOS_VENDOR}/directx-headers/include"
  "${BUNOS_VENDOR}/directx-headers/include/directx"
  "${BUNOS_VENDOR}/wil/include"
  "${BUNOS_CPPWINRT}"
  "${CMAKE_CURRENT_SOURCE_DIR}/../include")
target_compile_definitions(bunos_win_cpp PRIVATE BUNOS_WIN_CPP_BUILD)
target_compile_options(bunos_win_cpp PRIVATE /W4 /bigobj)
target_link_libraries(bunos_win_cpp PRIVATE d3d12 dxgi runtimeobject shell32 ole32)
